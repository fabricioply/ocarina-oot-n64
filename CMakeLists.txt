cmake_minimum_required(VERSION 3.26)
project(OcarinaN64 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_subdirectory(shared/real_instrument)
add_executable(test_n64_adapter tests/test_n64_adapter.cpp)
target_include_directories(test_n64_adapter PRIVATE src shared/real_instrument/tests)
target_link_libraries(test_n64_adapter PRIVATE real_instrument)
add_test(NAME n64_adapter COMMAND test_n64_adapter)
